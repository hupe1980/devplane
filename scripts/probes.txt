# The named probes a changelog row may point at.
#
# `scripts/changelog-ledger.txt` disposes of every rule-relevant row the vendor
# announces. A row marked `probe` names one of these by id, and that is what
# makes the third floor — `policy::ROWS_MEASURED_THROUGH` — mean something a
# person can re-establish rather than remember.
#
# The distinction this file exists to enforce:
#
#   a Rust test over our own matcher   proves Devplane behaves as *we* read the
#                                      row. It says nothing about the vendor.
#   a probe in here                    is a call the running vendor is asked
#                                      about, so a disagreement is findable.
#
# Both are real. Only the second can advance the measured floor, and the ledger
# said `case` for both until 2026-09-18 — which is how eleven rows came to claim
# in prose that they were "measured" with nothing able to re-establish it.
#
# Format, one probe per line:
#
#   <id> | <axis> | <rule> | <call>
#
# `<id>` matches [a-z0-9][a-z0-9_-]{2,63} and is what the ledger cites.
# `<axis>` is allow or deny. `<call>` is everything after the third `|`, so it
# may contain pipes. The rule and the call must both already exist in
# `verify-permissions-diff.sh`'s tables — a probe here that names a shape the
# harness cannot run is an error, not a silent skip.

tee-write-target        | allow | Bash(tee:*) | echo hi | tee ran.txt
assign-hides-substitution | allow | Bash(git config:*) | DIRSTACKSIZE=$(id) git config --local vp.ran 1
evaluated-assignment    | allow | Bash(git config:*) | OPTIND=1/0 git config --local vp.ran 1
loop-assigns-evaluated-var | allow | Bash(git config:*) | for OPTIND in 1 2; do git config --local vp.ran 1; done
subshell-unanalysable   | allow | Bash(git config:*) | (git config --local vp.ran 1)
analysis-barrier-eval   | deny  | Read(.env) | eval "cat .env"
read-deny-option-value  | deny  | Read(.env) | grep -f.env README.md
reader-fmt-path-deny    | deny  | Read(.env) | fmt .env > /dev/null
glob-operand-path-deny  | deny  | Read(.env) | cat .env* > /dev/null
glob-operand-bare-star  | deny  | Read(.env) | grep -l TOKEN * > /dev/null

# The first of the three dialect questions, and the only one askable with the
# machinery exactly as it stands: a **Bash** call under a **Read** rule.
#
# Every other git shape in the deny table names `.env` as a working-tree
# operand, which a path rule can see. `HEAD:.env` is a revision — the same
# bytes, reached without the rule's spelling appearing as a path at all. This
# matcher denies it; nothing had ever asked whether the vendor does.
git-show-revision       | deny  | Read(.env) | git show HEAD:.env
git-cat-file-revision   | deny  | Read(.env) | git cat-file -p HEAD:.env
